/-- under δ > 0 and L > 0 the two binder forms are the same proposition. -/
theorem probe_binders_equiv (δ L : ℝ) (hδ : 0 < δ) (hL : 0 < L) : (25 / L ≤ δ ↔ 25 / δ ≤ L) := by
  rw [div_le_iff₀ hL, div_le_iff₀ hδ, mul_comm]

/-- the assembly's hypotheses (0 < δ ≤ 1/2, 25/δ ≤ L) give every binder of `rstar_of_21L` except 21L ≤ t. -/
theorem probe_assembly_to_corollary (δ L : ℝ) (hδ0 : 0 < δ) (hδ : δ ≤ 1 / 2) (hL1 : 25 / δ ≤ L) :
    50 ≤ L ∧ 25 / L ≤ δ := by
  have h50 : (50 : ℝ) ≤ 25 / δ := by rw [le_div_iff₀ hδ0]; linarith
  have hL0 : 0 < L := by linarith
  exact ⟨by linarith, (probe_binders_equiv δ L hδ0 hL0).mpr hL1⟩

/-- the corollary's hypotheses (50 ≤ L, 25/L ≤ δ ≤ 1/2) give the assembly's δ-binders. -/
theorem probe_corollary_to_assembly (δ L : ℝ) (hL : 50 ≤ L) (hδ1 : 25 / L ≤ δ) :
    0 < δ ∧ 25 / δ ≤ L := by
  have hL0 : 0 < L := by linarith
  have hδ0 : 0 < δ := lt_of_lt_of_le (by positivity) hδ1
  exact ⟨hδ0, (probe_binders_equiv δ L hδ0 hL0).mp hδ1⟩

theorem probe_assembly_of_21L (C₁ t δ L : ℝ) (hC : 1 ≤ C₁) (ht : 3 ≤ t) (hδ0 : 0 < δ) (hδ : δ ≤ 1 / 2)
    (hL1 : 25 / δ ≤ L)
    (hL2 : 4 / δ * (Real.log (Real.log (3 + t)) + 2 * Real.log (1 / δ) + Real.log (2 * Separation.b1sym * C₁)) ≤ L)
    (h21 : 21 * L ≤ t)

    (hZ : ∀ ρ ∈ Z.carrier, |ρ.im - t| ≤ 73 * L → ρ ∉ Separation.orbit t δ → ρ.re = 1 / 2)
